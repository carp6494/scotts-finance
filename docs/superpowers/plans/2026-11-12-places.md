# Places and Addresses Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Every transaction shows where the money went, from Plaid's store location, a built-in headquarters table for online merchants, or what the user typed, with coordinates stored when known.

**Architecture:** New place columns on `transactions`, a `merchant_places` override table, a static headquarters dict in `backend/places_seed.py`, and `backend/places.py` that extracts, resolves in a fixed order, and fills rows the way `rules.categorize` does. The API returns a formatted `place_line` per row, accepts a place on PATCH, and reports coverage. The transaction list shows the line with an inline form.

**Tech Stack:** Python, FastAPI, sqlite3, pytest. React 19, Vite 7.

**Spec:** `docs/superpowers/specs/2026-11-12-places-design.md`. One adjustment to the spec: PATCH clears a place with `clear_place: true` rather than `place: null`, because pydantic cannot tell an omitted field from an explicit null.

**Conventions:** as in `docs/superpowers/plans/2026-11-12-categories-rules.md`. Tests from the project root: `.venv\Scripts\python -m pytest backend/tests -q`. Build in `frontend\`: `npm run build`. Commit trailer `Co-Authored-By: Claude Fable 5.1 <noreply@anthropic.com>`. Check the staged list before every commit. Never run `places.fill()` or any write against the real database outside the app. The headquarters dict is generic public knowledge; never add entries derived from the user's own transactions in a way that names their accounts.

```python
"""Where each transaction happened.

Resolution order for every row the user has not set by hand:
  1. a merchant_places override for the merchant (source user)
  2. Plaid's location block when it has a city or coordinates (source plaid)
  3. the built-in headquarters table by website domain, else by merchant name (source hq)
  4. nothing (source none)
"""
import json
import logging
import re
from urllib.parse import urlparse

import db
from places_seed import HQ
from rules import merchant_key

log = logging.getLogger("places")
FIELDS = ("place_address", "place_city", "place_region", "place_postal", "place_country", "place_lat", "place_lon", "place_store")


def _norm(s):
    return re.sub(r"[^a-z0-9]+", "", (s or "").lower())


_HQ_BY_NAME = {_norm(v[0]): v for v in HQ.values()}


def domain_of(website):
    if not website:
        return None
    w = str(website).strip().lower()
    if "://" in w:
        w = urlparse(w).netloc
    w = w.split("/")[0].split(":")[0]
    if w.startswith("www."):
        w = w[4:]
    return w or None


def extract(raw):
    """Plaid's location block as a dict, or None when it has neither a city nor coordinates."""
    try:
        r = json.loads(raw or "{}")
    except (TypeError, ValueError):
        return None
    if not isinstance(r, dict):
        return None
    loc = r.get("location") or {}
    out = {k: loc.get(k) for k in ("address", "city", "region", "postal_code", "country", "lat", "lon", "store_number")}
    if not (out["city"] or (out["lat"] is not None and out["lon"] is not None)):
        return None
    return out


def _website(t, raw):
    if t.get("website"):
        return t["website"]
    try:
        cps = (json.loads(raw or "{}") or {}).get("counterparties") or []
    except (TypeError, ValueError, AttributeError):
        return None
    return cps[0].get("website") if cps and isinstance(cps[0], dict) else None


def _hq(t, raw):
    dom = domain_of(_website(t, raw))
    if dom and dom in HQ:
        return HQ[dom]
    key = _norm(t.get("merchant_name"))
    return _HQ_BY_NAME.get(key) if key else None


def resolve(t, overrides):
    """Return (values dict keyed by FIELDS, source)."""
    mk = merchant_key(t)
    o = overrides.get(mk) if mk else None
    if o:
        return {"place_address": o["address"], "place_city": o["city"], "place_region": o["region"], "place_postal": o["postal"],
                "place_country": o["country"], "place_lat": o["lat"], "place_lon": o["lon"], "place_store": None}, "user"
    raw = t.get("raw")
    p = extract(raw)
    if p:
        return {"place_address": p["address"], "place_city": p["city"], "place_region": p["region"], "place_postal": p["postal_code"],
                "place_country": p["country"], "place_lat": p["lat"], "place_lon": p["lon"], "place_store": p["store_number"]}, "plaid"
    h = _hq(t, raw)
    if h:
        name, city, region, country, lat, lon = h
        return {"place_address": None, "place_city": city, "place_region": region or None, "place_postal": None,
                "place_country": country, "place_lat": lat, "place_lon": lon, "place_store": None}, "hq"
    return {k: None for k in FIELDS}, "none"


def _rows(transaction_ids):
    cols = "transaction_id, name, merchant_name, website, raw, place_source, " + ", ".join(FIELDS)
    base = f"SELECT {cols} FROM transactions WHERE COALESCE(place_source,'') != 'user'"
    if transaction_ids is None:
        return db.q(base)
    ids = list(dict.fromkeys(transaction_ids))
    rows = []
    for i in range(0, len(ids), 500):
        chunk = ids[i:i + 500]
        rows += db.q(f"{base} AND transaction_id IN ({','.join('?' * len(chunk))})", chunk)
    return rows


def fill(transaction_ids=None) -> int:
    """Resolve and write places for non-user rows. Returns how many rows changed."""
    overrides = {m["merchant_key"]: m for m in db.q("SELECT * FROM merchant_places")}
    updates = []
    for t in _rows(transaction_ids):
        vals, source = resolve(t, overrides)
        before = tuple(t.get(k) for k in FIELDS) + (t.get("place_source"),)
        after = tuple(vals[k] for k in FIELDS) + (source,)
        if before != after:
            updates.append(tuple(vals[k] for k in FIELDS) + (source, t["transaction_id"]))
    if updates:
        db.many(f"UPDATE transactions SET {', '.join(k + '=?' for k in FIELDS)}, place_source=? "
                "WHERE transaction_id=? AND COALESCE(place_source,'') != 'user'", updates)
    log.info("placed %s rows", len(updates))
    return len(updates)


def missing_ids():
    return [r["transaction_id"] for r in db.q("SELECT transaction_id FROM transactions WHERE place_source IS NULL")]


def _merchant_rows(transaction_id):
    t = db.one("SELECT merchant_name, name FROM transactions WHERE transaction_id=?", (transaction_id,))
    if not t:
        return None, []
    key = merchant_key(t)
    if not key:
        return None, [transaction_id]
    rows = db.q("SELECT transaction_id, merchant_name, name FROM transactions")
    return key, [r["transaction_id"] for r in rows if merchant_key(r) == key]


def set_user_place(transaction_id, place: dict):
    """Store the user's place for this merchant and apply it to every row of the merchant."""
    key, ids = _merchant_rows(transaction_id)
    if not ids:
        return 0
    vals = {k: (place.get(k) or "").strip() or None for k in ("address", "city", "region", "postal", "country")}
    lat, lon = place.get("lat"), place.get("lon")
    if key:
        db.run("""INSERT INTO merchant_places (merchant_key, name, address, city, region, postal, country, lat, lon, kind, updated_at)
                  VALUES (?,?,?,?,?,?,?,?,?,'user', datetime('now'))
                  ON CONFLICT(merchant_key) DO UPDATE SET address=excluded.address, city=excluded.city, region=excluded.region,
                  postal=excluded.postal, country=excluded.country, lat=excluded.lat, lon=excluded.lon, updated_at=datetime('now')""",
               (key, key, vals["address"], vals["city"], vals["region"], vals["postal"], vals["country"], lat, lon))
    for i in range(0, len(ids), 500):
        chunk = ids[i:i + 500]
        db.run(f"""UPDATE transactions SET place_address=?, place_city=?, place_region=?, place_postal=?, place_country=?,
                   place_lat=?, place_lon=?, place_store=NULL, place_source='user'
                   WHERE transaction_id IN ({','.join('?' * len(chunk))})""",
               (vals["address"], vals["city"], vals["region"], vals["postal"], vals["country"], lat, lon, *chunk))
    return len(ids)


def clear_user_place(transaction_id):
    """Remove the merchant override and re-resolve every row of the merchant."""
    key, ids = _merchant_rows(transaction_id)
    if key:
        db.run("DELETE FROM merchant_places WHERE merchant_key=?", (key,))
    for i in range(0, len(ids), 500):
        chunk = ids[i:i + 500]
        db.run(f"UPDATE transactions SET place_source=NULL WHERE transaction_id IN ({','.join('?' * len(chunk))})", chunk)
    return fill(ids)


def line(r) -> str:
    """One readable line for the UI."""
    src = r.get("place_source")
    if src in (None, "none"):
        return ""
    city = ", ".join(x for x in (r.get("place_city"), r.get("place_region")) if x)
    if src == "hq":
        return f"{city} · headquarters" if city else ""
    addr = r.get("place_address")
    return ", ".join(x for x in (addr, city) if x)


def coverage():
    out = {"plaid": 0, "hq": 0, "user": 0, "none": 0, "total": 0}
    for r in db.q("SELECT COALESCE(place_source,'none') AS s, COUNT(*) AS n FROM transactions GROUP BY 1"):
        out[r["s"] if r["s"] in out else "none"] += r["n"]
        out["total"] += r["n"]
    return out
```

import { useState } from 'react'
import { api, toast } from '../api'

const inp = { height: 30, borderRadius: 8, border: '1px solid var(--line)', background: 'var(--bg)', color: 'var(--text)', padding: '0 8px', fontSize: 12 }
const btn = { height: 30, padding: '0 10px', borderRadius: 8, border: 0, background: 'var(--accent)', color: '#fff', fontWeight: 600, fontSize: 12 }

// The address line under a transaction. Click to set or clear a place for this merchant.
export default function PlaceLine({ t, onSaved }) {
  const [open, setOpen] = useState(false)
  const [busy, setBusy] = useState(false)
  const [d, setD] = useState({ address: '', city: '', region: '', postal: '' })
  const icon = t.place_source === 'hq' ? '🏢' : t.place_source === 'user' ? '📌' : '📍'
  const text = t.place_line || 'Add a place'

  const start = (e) => {
    e.stopPropagation()
    setD({ address: t.place_address || '', city: t.place_city || '', region: t.place_region || '', postal: t.place_postal || '' })
    setOpen(true)
  }
  const save = async (e) => {
    e.preventDefault(); setBusy(true)
    try { const r = await api.patchTransaction(t.transaction_id, { place: d }); setOpen(false); onSaved(r) } catch (x) { toast(x.message) } finally { setBusy(false) }
  }
  const clear = async () => {
    setBusy(true)
    try { const r = await api.patchTransaction(t.transaction_id, { clear_place: true }); setOpen(false); onSaved(r) } catch (x) { toast(x.message) } finally { setBusy(false) }
  }

  if (!open) {
    return (
      <button type="button" onClick={start} title="Set the place for this merchant"
        style={{ border: 0, background: 'transparent', padding: 0, font: 'inherit', fontSize: 12, color: t.place_line ? 'var(--muted)' : 'var(--muted2)', cursor: 'pointer', textAlign: 'left' }}>
        {icon} {text}
      </button>
    )
  }
  return (
    <form onSubmit={save} onClick={(e) => e.stopPropagation()} style={{ display: 'flex', flexWrap: 'wrap', gap: 6, alignItems: 'center', marginTop: 4 }}>
      <input value={d.address} onChange={(e) => setD({ ...d, address: e.target.value })} placeholder="Street" style={{ ...inp, flex: '2 1 160px' }} />
      <input value={d.city} onChange={(e) => setD({ ...d, city: e.target.value })} placeholder="City" style={{ ...inp, flex: '1 1 110px' }} />
      <input value={d.region} onChange={(e) => setD({ ...d, region: e.target.value })} placeholder="State" style={{ ...inp, width: 64 }} />
      <input value={d.postal} onChange={(e) => setD({ ...d, postal: e.target.value })} placeholder="Zip" style={{ ...inp, width: 80 }} />
      <button type="submit" disabled={busy} className="press" style={btn}>Save for this merchant</button>
      {t.place_source === 'user' && <button type="button" disabled={busy} onClick={clear} style={{ ...btn, background: 'transparent', border: '1px solid var(--line)', color: 'var(--text2)', fontWeight: 500 }}>Clear</button>}
      <button type="button" onClick={() => setOpen(false)} style={{ ...btn, background: 'transparent', border: '1px solid var(--line)', color: 'var(--text2)', fontWeight: 500 }}>Cancel</button>
    </form>
  )
}
